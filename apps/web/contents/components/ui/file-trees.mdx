```tsx
import { FileTree } from "nextguide/blocks"

<FileTree title="NextGuide project" items={[{
  name:"my-docs",kind:"folder",children:[
    {name:"content",kind:"folder",children:[{name:"docs",kind:"folder",children:[{name:"index.mdx",kind:"code"},{name:"getting-started.mdx",kind:"code"}]}]},
    {name:"nextguide.config.json",kind:"json"},
    {name:"src",kind:"folder",children:[{name:"app",kind:"folder",children:[{name:"docs",kind:"folder",children:[{name:"layout.tsx",kind:"code"},{name:"[[...slug]]",kind:"folder",children:[{name:"page.tsx",kind:"code"}]}]}]}]}
  ]
}]} />
```

Nodes accept `name`, optional `kind`, `detail` and nested `children`. Folders expand with native keyboard-accessible `<details>` markup.
